cat << 'EOF' > run_benchmark.sh
#!/usr/bin/env bash
set -e

if [ -d "/sys/block/sdb" ]; then
    DEV="sdb"
elif [ -d "/sys/block/sda" ]; then
    DEV="sda"
else
    echo "No block device found."
    exit 1
fi

SCHEDULERS=("none" "mq-deadline" "kyber")
TEST_FILE="./test_payload.bin"
FILE_SIZE="1G"
RUNTIME=30

echo "=========================================================="
echo " Running on Host: $(hostname) | Device: /dev/$DEV"
echo " Workload: Mixed 70% Read / 30% Write (4KB)"
echo "=========================================================="

for SCHED in "${SCHEDULERS[@]}"; do
    echo ""
    echo "[+] Testing Scheduler: $SCHED"
    echo "$SCHED" | sudo tee /sys/block/$DEV/queue/scheduler > /dev/null
    echo "    Active: $(cat /sys/block/$DEV/queue/scheduler)"
    
    sync
    echo 3 | sudo tee /proc/sys/vm/drop_caches > /dev/null

    echo "    Executing fio (Runtime: ${RUNTIME}s)..."
    fio --name="bench_${SCHED}" \
        --filename="$TEST_FILE" \
        --size="$FILE_SIZE" \
        --rw=randrw \
        --rwmixread=70 \
        --bs=4k \
        --ioengine=libaio \
        --iodepth=16 \
        --direct=1 \
        --runtime="$RUNTIME" \
        --time_based \
        --group_reporting \
        --output-format=json \
        --output="result_${SCHED}.json"

    rm -f "$TEST_FILE"
    sleep 2
done

echo ""
echo "==================== FINAL BENCHMARK RESULTS ===================="
printf "%-14s | %-12s | %-12s | %-14s | %-14s\n" "Scheduler" "Read IOPS" "Write IOPS" "Read Lat (ms)" "Write Lat (ms)"
echo "----------------------------------------------------------------------------------"

for SCHED in "${SCHEDULERS[@]}"; do
    FILE="result_${SCHED}.json"
    if [ -f "$FILE" ]; then
        R_IOPS=$(jq '.jobs[0].read.iops' "$FILE")
        W_IOPS=$(jq '.jobs[0].write.iops' "$FILE")
        R_LAT=$(jq '.jobs[0].read.lat_ns.mean / 1000000' "$FILE")
        W_LAT=$(jq '.jobs[0].write.lat_ns.mean / 1000000' "$FILE")
        printf "%-14s | %-12.2f | %-12.2f | %-14.3f | %-14.3f\n" "$SCHED" "$R_IOPS" "$W_IOPS" "$R_LAT" "$W_LAT"
    fi
done
echo "=================================================================="
EOF
